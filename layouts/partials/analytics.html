{{- /* 说明：Analytics provider 分发入口。 用途：根据 `params.analytics.provider` 渲染对应 provider 的脚本注入。 约定： - 推荐：显式设置 `params.analytics.provider` + `params.analytics.providers.`。 - 若未设置 `provider` 且仅配置了 1 个 `providers`，会自动选用该 provider。 */ -}} {{- with site.Params.analytics -}} {{- $analytics := . -}} {{- $provider := "" -}} {{- with $analytics.provider -}} {{- $provider = lower . -}} {{- end -}} {{- $providers := $analytics.providers | default dict -}} {{- if and (not $provider) (eq (len $providers) 1) -}} {{- range $name, $_ := $providers -}} {{- if not $provider -}} {{- $provider = lower $name -}} {{- end -}} {{- end -}} {{- end -}} {{- if $provider -}} {{- $partial := printf "analytics/providers/%s.html" $provider -}} {{- $config := index $providers $provider -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- partial $partial (dict "Config" $config "Context" $) -}} {{- end -}} {{- end -}} {{- end -}}
