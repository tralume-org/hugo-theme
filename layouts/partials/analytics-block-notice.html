{{- /* 说明：Analytics provider 分发入口（拦截检测提示）。 用途：根据 `params.analytics.provider` 渲染对应 provider 的占位提示 UI。 约定：同 `partials/analytics.html` 的 provider 推断与配置读取策略。 */ -}} {{- with site.Params.analytics -}} {{- $analytics := . -}} {{- $provider := "" -}} {{- with $analytics.provider -}} {{- $provider = lower . -}} {{- end -}} {{- $providers := $analytics.providers | default dict -}} {{- if and (not $provider) (eq (len $providers) 1) -}} {{- range $name, $_ := $providers -}} {{- if not $provider -}} {{- $provider = lower $name -}} {{- end -}} {{- end -}} {{- end -}} {{- if $provider -}} {{- $partial := printf "analytics/providers/%s-block-notice.html" $provider -}} {{- $config := index $providers $provider -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- partial $partial (dict "Config" $config "Context" $) -}} {{- end -}} {{- end -}} {{- end -}}
