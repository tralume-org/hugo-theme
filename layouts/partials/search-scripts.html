{{- /* 说明：搜索 provider 脚本分发入口。 用途：根据 params.search.provider 加载对应 provider 的依赖脚本。 当前支持的 provider： - pagefind: 加载 Pagefind 生成的 /pagefind/pagefind.js - meilisearch: 直接通过 REST API 查询，无需额外脚本 */ -}} {{- with site.Params.search -}} {{- if .enable -}} {{- $provider := lower (default "pagefind" .provider) -}} {{- if eq $provider "pagefind" -}} {{- /* 说明：按语言分隔索引，pagefind.js 路径为 /pagefind//pagefind.js。 Pagefind 通过 import.meta.url 自动检测自身路径并定位分片文件。 */ -}} {{- $lang := lower site.Language.Lang -}} {{- $pagefindJS := printf "/pagefind/%s/pagefind.js" $lang | relURL }} {{- else if eq $provider "meilisearch" -}} {{- /* 说明：Meilisearch provider 使用 assets/js/search-providers/meilisearch.js 内的 fetch 实现。 这里仅做配置完整性提示，避免无声失败。 */ -}} {{- $meili := .meilisearch -}} {{- if not (and $meili.host $meili.indexUid) -}} {{- warnf "Search provider 'meilisearch' requires params.search.meilisearch.host and indexUid." -}} {{- end -}} {{- else -}} {{- warnf "Unknown search provider '%s'. No search scripts loaded." $provider -}} {{- end -}} {{- end -}} {{- end -}}
